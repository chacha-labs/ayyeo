import fs from 'node:fs'
import path from 'node:path'
import { isDeepStrictEqual } from 'node:util'
import { geminiAdapter } from '../packages/provider-gemini/src/adapter.ts'
import { claudeAdapter } from '../packages/provider-claude/src/adapter.ts'
import { compileQueryClassifier, determineAnswerMentioned, effectiveBrandNames, effectiveDomains } from '../packages/contracts/src/index.ts'
import { determineCitationState } from '../packages/canonry/src/citation-utils.ts'

// Offline only: normalize saved response bodies; never dispatch a query.
const artifactDir = process.argv[2]
if (!artifactDir) throw new Error('Usage: node --import tsx .context/verify-baseline.mjs <artifact-dir>')
const read = name => JSON.parse(fs.readFileSync(path.join(artifactDir, name), 'utf8'))
const save = (name, data) => fs.writeFileSync(path.join(artifactDir, name), JSON.stringify(data, null, 2) + '\n')
const manifest = read('manifest.json')
const project = read('project.json')
const snapshots = read('stored-snapshots.json')
const queryById = new Map(read('queries.json').map(query => [query.id, query.query]))
const names = effectiveBrandNames(project)
const domains = effectiveDomains(project)
const classifier = compileQueryClassifier(names)
const adapters = { gemini: geminiAdapter, claude: claudeAdapter }
const decode = value => typeof value === 'string' ? JSON.parse(value) : value
const fixtures = []
const mismatches = []
const observations = []
const identities = new Set()
for (const snapshot of snapshots) {
  const envelope = snapshot.raw_response
  const input = {
    provider: snapshot.provider,
    rawResponse: envelope.apiResponse,
    model: snapshot.model,
    servedModel: snapshot.served_model ?? undefined,
    groundingSources: envelope.groundingSources,
    searchQueries: envelope.searchQueries,
    retrievalStatus: snapshot.retrieval_status,
    retrievalContract: snapshot.retrieval_contract,
  }
  const normalized = adapters[snapshot.provider].normalizeResult(input)
  const expected = {
    answerText: snapshot.answer_text,
    citedDomains: decode(snapshot.cited_domains),
    groundingSources: envelope.groundingSources,
    searchQueries: envelope.searchQueries,
    retrievalStatus: snapshot.retrieval_status,
    citationState: snapshot.citation_state,
    answerMentioned: Boolean(snapshot.answer_mentioned),
  }
  const actual = {
    ...normalized,
    citationState: determineCitationState(normalized, domains),
    answerMentioned: determineAnswerMentioned(normalized.answerText, names, domains),
  }
  for (const [field, value] of Object.entries(expected)) {
    if (!isDeepStrictEqual(value, actual[field])) mismatches.push({ snapshotId: snapshot.id, provider: snapshot.provider, field })
  }
  const query = snapshot.query_text ?? queryById.get(snapshot.query_id)
  const identity = JSON.stringify([snapshot.provider, snapshot.model, query, snapshot.location])
  if (identities.has(identity)) mismatches.push({ snapshotId: snapshot.id, field: 'duplicate-observation' })
  identities.add(identity)
  if (!manifest.queries.some(item => item.text === query)) mismatches.push({ snapshotId: snapshot.id, field: 'unexpected-query' })
  if (manifest.providers[snapshot.provider]?.model !== snapshot.model) mismatches.push({ snapshotId: snapshot.id, field: 'requested-model-drift' })
  fixtures.push({ snapshotId: snapshot.id, query, queryClass: classifier.classify(query), input, expected })
  observations.push({ snapshotId: snapshot.id, provider: snapshot.provider, query,
    queryClass: classifier.classify(query), model: snapshot.model, servedModel: snapshot.served_model,
    mentioned: expected.answerMentioned, cited: snapshot.citation_state === 'cited',
    groundingSourceCount: normalized.groundingSources.length,
    retrievalStatus: snapshot.retrieval_status, retrievalContract: snapshot.retrieval_contract,
    captureStatus: snapshot.capture_status,
    answerText: normalized.answerText, groundingSources: normalized.groundingSources,
    humanReview: { reviewed: false, mentionCorrect: null, citationCorrect: null, sourceSupportCorrect: null } })
}
save('replay-fixtures.json', fixtures)
save('review-observations.json', observations)
const result = { snapshotCount: snapshots.length, assertionsPerSnapshot: 7, mismatches,
  replayPassed: snapshots.length > 0 && mismatches.length === 0,
  completeBasket: Object.keys(manifest.providers).every(provider => snapshots.filter(row => row.provider === provider).length === manifest.queries.length),
  note: 'Round-trip self-consistency only; fixtures have not been independently human-labeled.' }
save('replay-verification.json', result)
console.log(JSON.stringify(result))
if (!result.replayPassed) process.exitCode = 1
