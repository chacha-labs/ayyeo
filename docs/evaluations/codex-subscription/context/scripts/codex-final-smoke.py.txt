import json
import os
from pathlib import Path
import signal
import sqlite3
import subprocess
import threading
import time
import urllib.request

root = Path.cwd()
session = json.loads((root / '.context/codex-acceptance-session.json').read_text())
config_path = Path(session['configDir']) / 'config.yaml'
config = json.loads(subprocess.check_output(['node', '--input-type=module', '-e', "import fs from 'node:fs'; import {parse} from 'yaml'; process.stdout.write(JSON.stringify(parse(fs.readFileSync(process.argv[1], 'utf8'))));", str(config_path)], text=True))
out = root / '.context/codex-final-smoke'
out.mkdir(exist_ok=True)
secrets = [config['apiKey'], *(p.get('apiKey', '') for p in config['providers'].values())]
env = {k: v for k, v in os.environ.items() if k in ['PATH', 'HOME', 'TMPDIR', 'CODEX_HOME', 'LANG']}
env.update(CANONRY_CONFIG_DIR=session['configDir'], CANONRY_PORT=str(session['port']), CANONRY_AGENT_DISABLED='1', DO_NOT_TRACK='1', CANONRY_DISABLE_UPDATE_CHECK='1')
base = ['node', str(root / 'packages/canonry/dist/cli.js')]

def redact(text):
    for key in secrets:
        if key: text = text.replace(key, '[REDACTED]')
    return text

def save(name, value): (out / name).write_text(redact(json.dumps(value, indent=2)) + '\n')

def cli(args):
    result = subprocess.run(base + args + ['--format', 'json'], env=env, text=True, capture_output=True, timeout=90)
    if result.returncode: raise RuntimeError(redact(result.stderr))
    return json.loads(result.stdout)

if (out / 'result.json').exists(): raise RuntimeError('Smoke result exists; refusing duplicate spend.')
process = subprocess.Popen(base + ['serve'], env=env, stdout=subprocess.PIPE, stderr=subprocess.STDOUT, text=True, start_new_session=True)
def logs():
    with (out / 'server.log').open('w') as log:
        for line in process.stdout: log.write(redact(line)); log.flush()
thread = threading.Thread(target=logs, daemon=True); thread.start()
try:
    for _ in range(60):
        try:
            with urllib.request.urlopen(config['apiUrl'] + '/health', timeout=2) as response: health = json.load(response)
            break
        except OSError: time.sleep(1)
    else: raise RuntimeError('Server not ready')
    save('health.json', health)
    save('restored-status.json', cli(['settings', 'codex', 'status']))
    save('refreshed-status.json', cli(['settings', 'codex', 'refresh']))
    created = cli(['run', 'acceptance-aurora-solar', '--provider', 'codex', '--query', 'what is automated solar proposal software', '--probe', '--no-location'])
    for _ in range(50):
        result = cli(['run', 'show', created['id']])
        if result['status'] in ['completed', 'partial', 'failed', 'cancelled']: break
        time.sleep(5)
    else: raise RuntimeError('Smoke deadline exceeded')
    save('result.json', result)
    database = sqlite3.connect(f"file:{config['database']}?mode=ro", uri=True); database.row_factory = sqlite3.Row
    snapshots = [dict(row) for row in database.execute('SELECT * FROM query_snapshots WHERE run_id=?', (created['id'],))]
    for row in snapshots: row['raw_response'] = json.loads(row['raw_response'])
    save('stored-snapshots.json', snapshots); database.close()
    save('disconnected-status.json', cli(['settings', 'codex', 'disconnect']))
    auth = subprocess.run(['codex', 'login', 'status'], capture_output=True, text=True)
    still_signed_in = auth.returncode == 0 and 'chatgpt' in (auth.stdout + auth.stderr).lower()
    summary = {'status': result['status'], 'snapshots': len(snapshots), 'cliStillSubscriptionAuthenticated': still_signed_in,
               'verbatimWebPrograms': sum(len(row['raw_response']['apiResponse'].get('webToolCalls', [])) for row in snapshots)}
    save('summary.json', summary)
    print(json.dumps(summary), flush=True)
finally:
    if process.poll() is None:
        os.killpg(process.pid, signal.SIGTERM)
        try: process.wait(timeout=20)
        except subprocess.TimeoutExpired: os.killpg(process.pid, signal.SIGKILL); process.wait()
    thread.join(timeout=5)
    for file in out.iterdir():
        if file.is_file(): assert not any(key and key in file.read_text() for key in secrets), file.name
