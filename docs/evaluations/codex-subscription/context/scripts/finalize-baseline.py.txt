import hashlib
import json
import pathlib
import shutil
import socket
import subprocess

root = pathlib.Path(__file__).resolve().parent.parent
session = json.loads((root / '.context/baseline-session.json').read_text())
out = pathlib.Path(session['artifactDir'])
load = lambda name: json.loads((out / name).read_text())
manifest, summary, replay = load('manifest.json'), load('summary.json'), load('replay-verification.json')
observations = load('review-observations.json')
tests = load('test-summary.json')
query_rows = []
for item in manifest['queries']:
    matching = [row for row in observations if row['query'] == item['text']]
    cells = []
    for provider in manifest['providers']:
        observation = next((row for row in matching if row['provider'] == provider), None)
        if observation is None:
            cells.append('–')
        else:
            cells.append(('C' if observation['cited'] else 'c') + ('M' if observation['mentioned'] else 'm'))
    query_rows.append(f"| {item['key']} | {item['text']} | {matching[0]['queryClass'] if matching else 'unavailable'} | {' | '.join(cells)} |")
(out / 'observations.md').write_text('# Fixed basket observations\n\n'
    'Legend: **C** cited, **c** not cited; **M** mentioned, **m** not mentioned; **–** missing. '
    'Labels are the stored Canonry outputs; independent human review is pending.\n\n'
    '| ID | Query | Query class | Gemini | Claude |\n|---|---|---|---|---|\n' + '\n'.join(query_rows) + '\n')
rows = []
for provider, settings in manifest['providers'].items():
    captured = [row for row in observations if row['provider'] == provider]
    assert len({row['query'] for row in captured}) == len(captured)
    rows.append(f"| {provider} | `{settings['model']}` | {len(captured)}/20 | {sum(bool(row['answerText'].strip()) for row in captured)} |")
class_rows = []
for provider in manifest['providers']:
    for query_class in ['non-brand', 'branded']:
        group = [row for row in observations if row['provider'] == provider and row['queryClass'] == query_class]
        class_rows.append(f"| {provider} | {query_class} | {len(group)} | {sum(row['mentioned'] for row in group)} | {sum(row['cited'] for row in group)} |")
quality = {provider: {
    'emptyAnswers': sum(not row['answerText'].strip() for row in observations if row['provider'] == provider),
    'withoutGroundingSources': sum(row['groundingSourceCount'] == 0 for row in observations if row['provider'] == provider),
    'servedModels': sorted({row['servedModel'] for row in observations if row['provider'] == provider and row['servedModel']}),
    'retrievalStatuses': sorted({row['retrievalStatus'] for row in observations if row['provider'] == provider}),
    'captureStatuses': sorted({row['captureStatus'] for row in observations if row['provider'] == provider}),
} for provider in manifest['providers']}
(out / 'quality-summary.json').write_text(json.dumps(quality, indent=2) + '\n')
test_count = sum(result['numPassedTests'] for result in tests.values())
with socket.socket() as connection:
    stopped = connection.connect_ex(('127.0.0.1', session['port'])) != 0
assert stopped, 'Baseline listener is still active'
assert not subprocess.check_output(['git', 'status', '--porcelain'], cwd=root, text=True).strip(), 'Unexpected tracked changes'
status = 'COMPLETE' if summary['complete'] and replay['replayPassed'] else 'INCOMPLETE'
text = f"""# Aurora Solar provider baseline — {status}

Recorded from Canonry {session['version']}, commit `{session['commit']}` (clean checkout matching origin/main at capture).
Target: **aurorasolar.com**, brand **Aurora Solar**, US / English, no location override.
20 exact queries selected from the repository's B2B SaaS discovery fixture; two name Aurora Solar and 18 are non-brand relative to Aurora Solar. Exact inputs, source indices, model choices, quota settings, and basket hash are in [manifest.json](manifest.json).

## Capture

One pass per provider, using a one-query pilot and then the remaining 19 queries. No application-level repeat sweeps. Existing provider/SDK retries remain unchanged. All runs use `probe` in an isolated local database. No production data, scheduled jobs, or agent wakeups were used. The temporary server has been stopped.

| Provider | Requested model | Saved / expected | Non-empty answers |
|---|---|---|---|
{chr(10).join(rows)}

## Observed signals

Mention = brand/domain appears in answer text. Cited = project domain appears in source evidence. These are baseline observations, not independently judged answer quality or rankings. Branded and non-brand denominators remain separate.

| Provider | Query class | Observations | Mentioned | Cited |
|---|---|---|---|---|
{chr(10).join(class_rows)}

## Verification

- CLI/server build succeeded and `/health` commit matched the captured checkout.
- {test_count} focused tests passed: provider adapters, execution limits/retrieval handling, provider surface/location handling, and probe exclusion. Full workspace CI was not run for this capture.
- Offline replay: {replay['snapshotCount']} saved responses; {len(replay['mismatches'])} field mismatches against stored observations. This proves round-trip consistency, not independently labeled correctness.
- No duplicate query/provider observations; raw response bodies and normalized results are retained for later replay.
- Provider keys are outside this artifact directory in an isolated local config with mode 0600. Artifacts are scanned for the supplied keys and the local instance token.

## Files and comparison procedure

- [summary.json](summary.json): terminal statuses, run IDs, expected/actual counts, elapsed times.
- [provenance.json](provenance.json) and [health.json](health.json): git/build/runtime identity, lockfile and built-file hashes.
- [results-export.json](results-export.json): standard Canonry export, including probes.
- [stored-snapshots.json](stored-snapshots.json): saved database observations with raw provider bodies; read-only extraction from the isolated database.
- [replay-fixtures.json](replay-fixtures.json): frozen raw inputs plus baseline expected normalized fields.
- [replay-verification.json](replay-verification.json): baseline replay result.
- [review-observations.json](review-observations.json): answers, sources, signal labels, and unfilled human-review fields.
- [observations.md](observations.md): compact per-query citation + mention table for both providers.
- [quality-summary.json](quality-summary.json): empty answers, source availability, served models, retrieval/capture statuses.
- [test-summary.json](test-summary.json): local checks; detailed test outputs and sanitized run logs are adjacent.

For an after-change capture, use a fresh isolated instance and this exact manifest, retain the before fixtures unchanged, and record the new server commit. Compare deterministic normalized fields against the frozen `expected` objects in `replay-fixtures.json`; live answer differences must be reviewed separately. The archived scripts document how this baseline was captured; their active copies live at repository `.context/capture-baseline.py` and `.context/verify-baseline.mjs`. The replay helper writes output files, so run it against a copy when checking another revision.

## Limits

This is one 20-query pass per provider, not the planned three-repeat stability study. No Codex results exist yet. This captures the Simple portfolio path; Advanced portfolio parity remains a later integration test. Human/source-support review is still pending. `unknown` retrieval status is retained rather than inferred from source counts. Raw responses alone do not preserve the exact future content of third-party pages.
"""
(out / 'README.md').write_text(text)
config = json.loads((pathlib.Path(session['configDir']) / 'config.yaml').read_text())
secrets = [config['apiKey'], *(entry['apiKey'] for entry in config['providers'].values())]
for name in ['capture-baseline.py', 'verify-baseline.mjs', 'finalize-baseline.py']:
    shutil.copyfile(root / '.context' / name, out / name)
for artifact in out.iterdir():
    if artifact.is_file():
        body = artifact.read_text()
        assert not any(secret in body for secret in secrets), f'Credential leak: {artifact.name}'
hashes = [f'{hashlib.sha256(file.read_bytes()).hexdigest()}  {file.name}' for file in sorted(out.iterdir()) if file.is_file() and file.name != 'SHA256SUMS']
(out / 'SHA256SUMS').write_text('\n'.join(hashes) + '\n')
print(json.dumps({'status': status, 'readme': str(out / 'README.md'), 'quality': quality, 'testsPassed': test_count, 'credentialScan': 'passed', 'serverStopped': stopped}, indent=2))
