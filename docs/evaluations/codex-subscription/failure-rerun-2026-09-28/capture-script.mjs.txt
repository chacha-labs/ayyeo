import fs from 'node:fs'
import path from 'node:path'
import { execFileSync } from 'node:child_process'
import { CodexRuntime, CodexProcess, CODEX_INSTRUCTIONS } from './diagnostic-runtime-0.158.0.ts'
import { captureCodexSources, normalizeCodexEvidence } from '../packages/provider-codex/src/evidence.ts'
import { isVerbatimWebProgram } from '../packages/provider-codex/src/web-provenance.ts'
import { readMarkdownAnswer, describeError } from '../packages/contracts/src/index.ts'

const output = path.resolve(process.argv[2] ?? `.context/codex-failure-rerun-${new Date().toISOString().replaceAll(/[:.]/g, '-')}`)
fs.mkdirSync(output, { recursive: false, mode: 0o700 })
const basket = JSON.parse(fs.readFileSync('docs/evaluations/codex-subscription/before/manifest.json', 'utf8'))
const queries = ['q16', 'q15'].map(key => basket.queries.find(query => query.key === key))
const model = 'gpt-6-astra'
let active
const save = (name, value) => fs.writeFileSync(path.join(output, name), JSON.stringify(value, null, 2) + '\n', { mode: 0o600 })
const runtime = new CodexRuntime(cwd => {
  const process = new CodexProcess(cwd)
  process.subscribe((method, params) => {
    if (!active || !params || typeof params !== 'object') return
    const item = params.item
    let event
    if (method === 'item/completed' && item?.type === 'agentMessage' && item.phase === 'final_answer') {
      event = { method, threadId: params.threadId, item: { type: item.type, id: item.id, phase: item.phase, text: item.text } }
    } else if (method === 'item/completed' && item?.type === 'webSearch') {
      event = { method, threadId: params.threadId, item: { type: item.type, id: item.id, action: item.action, query: item.query } }
    } else if (method === 'rawResponseItem/completed' && ['custom_tool_call', 'function_call'].includes(item?.type) && item.name === 'exec') {
      event = { method, threadId: params.threadId, item: { type: item.type, name: item.name, call_id: item.call_id, input: item.input, arguments: item.arguments } }
    } else if (method === 'rawResponseItem/completed' && ['custom_tool_call_output', 'function_call_output'].includes(item?.type)) {
      event = { method, threadId: params.threadId, item: { type: item.type, call_id: item.call_id, output: item.output } }
    } else if (method === 'turn/completed') {
      event = { method, threadId: params.threadId, turn: { id: params.turn?.id, status: params.turn?.status, error: params.turn?.error } }
    }
    if (event) {
      active.events.push(event)
      fs.appendFileSync(path.join(output, `${active.key}-events.jsonl`), JSON.stringify(event) + '\n', { mode: 0o600 })
    }
  })
  return process
})
const results = []
try {
  const inspection = await runtime.inspect()
  save('manifest.json', { startedAt: new Date().toISOString(), commit: execFileSync('git', ['rev-parse', 'HEAD'], { encoding: 'utf8' }).trim(), model, runtimeVersion: inspection.runtimeVersion, subscriptionAuthenticated: true, prompt: CODEX_INSTRUCTIONS, queries, location: null, capture: 'Allowlisted final-answer and web-tool events written before production validation. No auth replies, reasoning, or input instruction events.', mode: 'Diagnostic-only copy of production runtime with version guard set to observed 0.158.0; all prompts, isolation, capture, and evidence validation unchanged. No project observations written.' })
  if (!inspection.models.some(entry => entry.id === model)) throw new Error('Pinned benchmark model unavailable')
  runtime.enable()
  for (const query of queries) {
    active = { key: query.key, events: [] }
    const started = Date.now()
    let error = null
    let result = null
    try {
      result = await runtime.execute({ query: query.text, canonicalDomains: ['aurorasolar.com'], competitorDomains: [], runId: `diagnostic-${query.key}` }, model, inspection.accountId)
    } catch (caught) { error = describeError(caught) }
    const { events } = active
    active = undefined
    const programs = new Map(events.filter(event => event.item?.name === 'exec').map(event => [event.item.call_id, event.item.input ?? event.item.arguments ?? '']))
    const calls = events.filter(event => ['custom_tool_call_output', 'function_call_output'].includes(event.item?.type) && programs.has(event.item.call_id)).map(event => ({ callId: event.item.call_id, program: programs.get(event.item.call_id), accepted: isVerbatimWebProgram(programs.get(event.item.call_id)), outputs: Array.isArray(event.item.output) ? event.item.output.flatMap(part => typeof part.text === 'string' ? [part.text] : []) : [] }))
    const answers = new Map(events.filter(event => event.item?.type === 'agentMessage').map(event => [event.item.id, event.item.text]))
    const answerText = [...answers.values()].join('\n\n')
    const sources = captureCodexSources(calls.filter(call => call.accepted).flatMap(call => call.outputs))
    const rendered = readMarkdownAnswer(answerText.replace(/\uE200cite\uE202[^\uE201]+\uE201/g, ''))
    const links = rendered.links.map(({ uri }) => ({ uri, matched: sources.some(source => new URL(source.uri).href === new URL(uri).href), sameHostCandidates: sources.filter(source => new URL(source.uri).hostname === new URL(uri).hostname) }))
    const references = [...answerText.matchAll(/\uE200cite\uE202([^\uE201]+)\uE201/g)].flatMap(match => match[1].split('\uE202')).map(reference => ({ reference, matched: sources.some(source => source.reference === reference) }))
    const diagnostic = { query, elapsedSeconds: (Date.now() - started) / 1000, accepted: result !== null, error, answerText, sources, links, references, webToolCalls: calls, normalized: result ? normalizeCodexEvidence(result) : null }
    save(`${query.key}-diagnostic.json`, diagnostic)
    const summary = { query: query.key, accepted: diagnostic.accepted, error, answerCharacters: answerText.length, sourceCount: sources.length, unmatchedLinks: links.filter(link => !link.matched).map(link => link.uri), rejectedPrograms: calls.filter(call => !call.accepted).length, elapsedSeconds: diagnostic.elapsedSeconds }
    results.push(summary)
    save('summary.json', results)
    console.log(JSON.stringify(summary))
    if (Date.now() - started < 10000) await new Promise(resolve => setTimeout(resolve, 10000 - (Date.now() - started)))
  }
} catch (error) {
  save('fatal.json', { error: describeError(error) })
  console.error(describeError(error))
  process.exitCode = 1
} finally {
  active = undefined
  await runtime.close()
  console.log(JSON.stringify({ output, runtimeClosed: true }))
}
